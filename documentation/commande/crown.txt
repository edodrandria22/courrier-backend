#!/bin/bash

DB_NAME="courrier"
DB_USER="mesupres"
BACKUP_DIR="/home/sysadmin/mesupres/backup"

DATE=$(date +"%Y-%m-%d_%H-%M-%S")
BACKUP_FILE="$BACKUP_DIR/${DB_NAME}_${DATE}.sql"

pg_dump -h 127.0.0.1 -U "$DB_USER" -d "$DB_NAME" > "$BACKUP_FILE"

if [ $? -eq 0 ]; then
    echo "[$(date)] Sauvegarde réussie : $BACKUP_FILE"
else
    echo "[$(date)] ERREUR lors de la sauvegarde"
    exit 1
fi

find "$BACKUP_DIR" -type f -name "${DB_NAME}_*.sql" -mtime +30 -delete
---
sudo chmod +x /usr/local/bin/backup_courrier.sh


sudo crontab -e
0 0 * * * /usr/local/bin/backup_courrier.sh >> /var/log/backup_courrier.log 2>&1
